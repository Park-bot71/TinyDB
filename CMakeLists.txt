cmake_minimum_required(VERSION 3.20)
project(tinydb VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(TINYDB_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(TINYDB_FETCH_GTEST "Download pinned GoogleTest if no installed package is found" OFF)
set(TINYDB_GTEST_SOURCE_DIR "" CACHE PATH "Optional local GoogleTest source directory")

find_package(Threads REQUIRED)
add_library(tinydb
  src/recovery/wal.cpp
  src/concurrency/lock_manager.cpp
  src/transaction/database.cpp
  src/index/index_page.cpp
  src/index/b_plus_tree.cpp
  src/buffer/buffer_pool.cpp
  src/storage/page.cpp
  src/storage/disk_manager.cpp
  src/storage/slotted_page.cpp
  src/storage/heap_file.cpp
  src/storage/record.cpp)
target_include_directories(tinydb PUBLIC include)
target_link_libraries(tinydb PUBLIC Threads::Threads)
target_compile_options(tinydb PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
if(TINYDB_SANITIZERS)
  target_compile_options(tinydb PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(tinydb PUBLIC -fsanitize=address,undefined)
endif()

add_executable(storage_demo examples/storage_demo.cpp)
target_link_libraries(storage_demo PRIVATE tinydb)
add_executable(index_demo examples/index_demo.cpp)
target_link_libraries(index_demo PRIVATE tinydb)
add_executable(transaction_demo examples/transaction_demo.cpp)
target_link_libraries(transaction_demo PRIVATE tinydb)
add_library(tinydb_cli_support src/cli/shell.cpp)
target_link_libraries(tinydb_cli_support PUBLIC tinydb)
target_compile_options(tinydb_cli_support PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
add_executable(tinydb_shell src/cli/main.cpp)
set_target_properties(tinydb_shell PROPERTIES OUTPUT_NAME tinydb)
target_link_libraries(tinydb_shell PRIVATE tinydb_cli_support)
target_compile_options(tinydb_shell PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
add_executable(tinydb_bench benchmarks/benchmark.cpp)
target_link_libraries(tinydb_bench PRIVATE tinydb)
target_compile_options(tinydb_bench PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
add_executable(tinydb_validation benchmarks/validation_suite.cpp)
target_link_libraries(tinydb_validation PRIVATE tinydb)
target_compile_options(tinydb_validation PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)

include(CTest)
if(BUILD_TESTING)
  if(TINYDB_GTEST_SOURCE_DIR)
    add_subdirectory("${TINYDB_GTEST_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/googletest" EXCLUDE_FROM_ALL)
  else()
    find_package(GTest CONFIG QUIET)
    if(NOT GTest_FOUND)
      if(NOT TINYDB_FETCH_GTEST)
        message(FATAL_ERROR "Install libgtest-dev, set TINYDB_GTEST_SOURCE_DIR, or enable TINYDB_FETCH_GTEST")
      endif()
      include(FetchContent)
      # Fixed upstream release commit; only used when explicitly requested.
      FetchContent_Declare(googletest
        GIT_REPOSITORY https://github.com/google/googletest.git
        GIT_TAG f8d7d77c06936315286eb55f8de22cd23c188571)
      FetchContent_MakeAvailable(googletest)
    endif()
  endif()
  add_executable(storage_tests tests/storage_tests.cpp)
  target_link_libraries(storage_tests PRIVATE tinydb GTest::gtest_main)
  target_compile_options(storage_tests PRIVATE -Wall -Wextra -Wpedantic)
  include(GoogleTest)
  gtest_discover_tests(storage_tests DISCOVERY_TIMEOUT 30 PROPERTIES TIMEOUT 30)
  add_executable(buffer_tests tests/buffer_tests.cpp)
  target_link_libraries(buffer_tests PRIVATE tinydb GTest::gtest_main)
  target_compile_options(buffer_tests PRIVATE -Wall -Wextra -Wpedantic)
  gtest_discover_tests(buffer_tests DISCOVERY_TIMEOUT 30 PROPERTIES TIMEOUT 30)
  add_executable(index_tests tests/index_tests.cpp)
  target_link_libraries(index_tests PRIVATE tinydb GTest::gtest_main)
  target_compile_options(index_tests PRIVATE -Wall -Wextra -Wpedantic)
  gtest_discover_tests(index_tests DISCOVERY_TIMEOUT 30 PROPERTIES TIMEOUT 120)
  add_executable(transaction_tests tests/transaction_tests.cpp)
  target_link_libraries(transaction_tests PRIVATE tinydb GTest::gtest_main)
  target_compile_options(transaction_tests PRIVATE -Wall -Wextra -Wpedantic)
  gtest_discover_tests(transaction_tests DISCOVERY_TIMEOUT 30 PROPERTIES TIMEOUT 60)
  add_executable(recovery_worker tests/recovery_worker.cpp)
  target_link_libraries(recovery_worker PRIVATE tinydb)
  add_executable(recovery_tests tests/recovery_tests.cpp)
  target_link_libraries(recovery_tests PRIVATE tinydb GTest::gtest_main)
  target_compile_options(recovery_tests PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_definitions(recovery_tests PRIVATE RECOVERY_WORKER_PATH="$<TARGET_FILE:recovery_worker>")
  add_dependencies(recovery_tests recovery_worker)
  gtest_discover_tests(recovery_tests DISCOVERY_TIMEOUT 30 PROPERTIES TIMEOUT 120)
  add_executable(cli_tests tests/cli_tests.cpp)
  target_link_libraries(cli_tests PRIVATE tinydb_cli_support GTest::gtest_main)
  target_compile_options(cli_tests PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_definitions(cli_tests PRIVATE TINYDB_SHELL_PATH="$<TARGET_FILE:tinydb_shell>")
  target_compile_definitions(cli_tests PRIVATE TINYDB_BENCH_PATH="$<TARGET_FILE:tinydb_bench>")
  add_dependencies(cli_tests tinydb_shell tinydb_bench)
  gtest_discover_tests(cli_tests DISCOVERY_TIMEOUT 30 PROPERTIES TIMEOUT 30)
  add_executable(stress_tests tests/stress_tests.cpp)
  target_link_libraries(stress_tests PRIVATE tinydb GTest::gtest_main)
  target_compile_options(stress_tests PRIVATE -Wall -Wextra -Wpedantic)
  gtest_discover_tests(stress_tests DISCOVERY_TIMEOUT 30 PROPERTIES TIMEOUT 120)
  add_executable(validation_tests tests/validation_tests.cpp)
  target_link_libraries(validation_tests PRIVATE tinydb GTest::gtest_main)
  target_compile_options(validation_tests PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_definitions(validation_tests PRIVATE VALIDATION_SUITE_PATH="$<TARGET_FILE:tinydb_validation>")
  add_dependencies(validation_tests tinydb_validation)
  gtest_discover_tests(validation_tests DISCOVERY_TIMEOUT 30 PROPERTIES TIMEOUT 120)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME BenchmarkDriverTests
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/benchmark_driver_tests.py" "$<TARGET_FILE:tinydb_validation>")
    set_tests_properties(BenchmarkDriverTests PROPERTIES TIMEOUT 60)
  endif()
endif()
